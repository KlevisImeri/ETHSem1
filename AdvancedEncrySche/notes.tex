\begin{definition}[KEM/DEM paradigm]
\[
\begin{aligned}
\mathsf{KEM}&:=(\mathsf{Gen},\mathsf{Encaps},\mathsf{Decaps}),\\
\mathsf{Gen}(1^\lambda)&\longrightarrow(pk,sk),\\
\mathsf{Encaps}(pk)&\longrightarrow(K,c_1),\\
\mathsf{Decaps}(sk,c_1)&\longrightarrow K'\ \text{or}\ \bot.
\end{aligned}
\qquad
\begin{aligned}
\mathsf{DEM}&:=(\mathsf{Enc},\mathsf{Dec}),\\
\mathsf{Enc}(K,m)&\longrightarrow c_2,\\
\mathsf{Dec}(K,c_2)&\longrightarrow m'\ \text{or}\ \bot.
\end{aligned}
\]
\[
\begin{aligned}
\text{Hybrid encryption}:\quad
&(K,c_1)\leftarrow\mathsf{KEM.Encaps}(pk),\quad
c_2\leftarrow\mathsf{DEM.Enc}(K,m),\quad ct:=(c_1,c_2),\\
\text{Hybrid decryption}:\quad
&K'\leftarrow\mathsf{KEM.Decaps}(sk,c_1),\quad
m'\leftarrow\mathsf{DEM.Dec}(K',c_2)
\quad(K'\ne\bot).
\end{aligned}
\]
\[
K'=\bot\ \Longrightarrow\ \text{hybrid decryption outputs }\bot.
\]
\[
\begin{aligned}
\mathsf{KEM}&:\text{ Key Encapsulation Mechanism},\\
\mathsf{DEM}&:\text{ Data Encapsulation Mechanism},\\
\lambda\in\mathbb{N}&:\text{ security parameter},\\
pk,sk&:\text{ public and secret keys},\\
K&:\text{ symmetric key},\\
m&:\text{ message},\\
c_1,c_2&:\text{ key encapsulation, encrypted data},\\
\bot&:\text{ failure}.
\end{aligned}
\]
\end{definition}

\section{Lecture 1}

\begin{definition}[Probabilistic polynomial-time algorithm]
\[
\mathrm{PPT}(A) \iff \exists p\in\poly\ \ \forall x\in\{0,1\}^{*}\ \ \forall r:\ \mathrm{time}_A(x;r)\le p(|x|).
\]
\end{definition}

\begin{definition}[Negligible function]
\[
\mu:\mathbb{N}\to\mathbb{R}\text{ negligible}
\ \iff\ 
\forall c\in\mathbb{N}\ \exists \lambda_0\in\mathbb{N}\ \forall \lambda\ge\lambda_0:
\quad |\mu(\lambda)|<\frac{1}{\lambda^c}.
\]
\end{definition}

\begin{problem}[Exercise sheet 1: Negligible Functions]
Prove or disprove that the following functions are negligible.
You can use that $f(n)=1/2^n$ is negligible.
\begin{enumerate}
\setcounter{enumi}{-1}
\item $f_0(\lambda)=\dfrac{1}{2^\lambda}$.
\begin{solution}
\[
\begin{aligned}
&\forall c\in\mathbb{N}:\quad
\lim_{\lambda\to\infty}\frac{\lambda^c}{2^\lambda}=0\\
\Longrightarrow\quad
&\forall c\in\mathbb{N}\ \exists\lambda_0\in\mathbb{N}\
\forall\lambda\ge\lambda_0:
\quad\frac{\lambda^c}{2^\lambda}<1\\
\Longrightarrow\quad
&\forall c\in\mathbb{N}\ \exists\lambda_0\in\mathbb{N}\
\forall\lambda\ge\lambda_0:
\quad |f_0(\lambda)|=\frac{1}{2^\lambda}<\frac{1}{\lambda^c}\\
\Longrightarrow\quad
&f_0\text{ is negligible}.
\end{aligned}
\]
\end{solution}
\item $f_1(n)=\dfrac{1}{n^{20}}$.
\begin{solution}
\[
\begin{aligned}
c:=20\in\mathbb{N}
\ \Longrightarrow\
&\forall\lambda_0\in\mathbb{N}\ \exists\lambda\ge\lambda_0:
\quad |f_1(\lambda)|=\frac{1}{\lambda^{20}}
=\frac{1}{\lambda^c}\ge\frac{1}{\lambda^c}\\
\ \Longrightarrow\
&\exists c\in\mathbb{N}\ \forall\lambda_0\in\mathbb{N}\
\exists\lambda\ge\lambda_0:
\quad |f_1(\lambda)|\ge\frac{1}{\lambda^c}\\
\ \Longrightarrow\
&f_1\text{ is non-negligible}.
\end{aligned}
\]
\end{solution}
\item $f_2(n)=\dfrac{2^n}{(2^n)^2}$.
\begin{solution}
\[
|f_2(\lambda)|
=\frac{2^\lambda}{(2^\lambda)^2}
=\frac{1}{2^\lambda}
=f_0(\lambda)
\ \Longrightarrow\quad f_2\text{ is negligible}.
\]
\end{solution}
\item $f_3(n)=\left|\dfrac{(-1)^n}{2^n}\right|$.
\begin{solution}
\[
|f_3(\lambda)|
=\left|\frac{(-1)^\lambda}{2^\lambda}\right|
=\frac{1}{2^\lambda}
=f_0(\lambda)
\ \Longrightarrow\quad f_3\text{ is negligible}.
\]
\end{solution}
\item $f_4(n)=\dfrac{(-1)^n}{2^n}$.
\begin{solution}
\[
|f_4(\lambda)|
=\left|\frac{(-1)^\lambda}{2^\lambda}\right|
=\frac{1}{2^\lambda}
=f_0(\lambda)
\ \Longrightarrow\quad f_4\text{ is negligible}.
\]
\end{solution}
\item $f_5(n)=\dfrac{n^2+n^7}{2^n}$.
\begin{solution}
\[
\begin{aligned}
&\forall c\in\mathbb{N}:\quad
\lim_{\lambda\to\infty}\lambda^c|f_5(\lambda)|
=\lim_{\lambda\to\infty}
\left(\frac{\lambda^{c+2}}{2^\lambda}
+\frac{\lambda^{c+7}}{2^\lambda}\right)=0\\
\Longrightarrow\quad
&\forall c\in\mathbb{N}\ \exists\lambda_0\in\mathbb{N}\
\forall\lambda\ge\lambda_0:
\quad\lambda^c|f_5(\lambda)|<1\\
\Longrightarrow\quad
&\forall c\in\mathbb{N}\ \exists\lambda_0\in\mathbb{N}\
\forall\lambda\ge\lambda_0:
\quad |f_5(\lambda)|<\frac{1}{\lambda^c}\\
\Longrightarrow\quad
&f_5\text{ is negligible}.
\end{aligned}
\]
\end{solution}
\end{enumerate}
\end{problem}

\begin{definition}[Public key encryption scheme]
\[
\Pi := (\mathsf{Gen},\mathsf{Enc},\mathsf{Dec}).
\]
\[
\mathrm{PKE}(\Pi)\iff
\left\{
\begin{aligned}
&\mathrm{PPT}(\mathsf{Gen})\land
\mathrm{PPT}(\mathsf{Enc})\land
\mathrm{PPT}(\mathsf{Dec}),\\[3pt]
&\forall\lambda\in\mathbb{N}:\\[2pt]
&\quad\mathcal{K}_{\mathrm{pub},\lambda}\times\mathcal{K}_{\mathrm{sec},\lambda}
\ni(pk,sk)\longleftarrow\mathsf{Gen}(1^\lambda),\\[3pt]
&\quad\begin{aligned}
\forall pk\in\mathcal{K}_{\mathrm{pub},\lambda}\ \forall m\in\mathcal{M}_\lambda:
&\quad\mathcal{C}_\lambda\ni ct\longleftarrow\mathsf{Enc}(pk,m),\\[3pt]
\forall sk\in\mathcal{K}_{\mathrm{sec},\lambda}\ \forall ct\in\mathcal{C}_\lambda:
&\quad\mathcal{M}_\lambda\cup\{\bot\}\ni m'\longleftarrow\mathsf{Dec}(sk,ct).
\end{aligned}
\end{aligned}
\right.
\]
\[
\begin{aligned}
\mathsf{Gen},\mathsf{Enc},\mathsf{Dec}&:\text{ key generation, encryption, decryption algorithms},\\
\lambda&:\text{ security parameter},\\
\mathcal{K}_{\mathrm{pub},\lambda},\mathcal{K}_{\mathrm{sec},\lambda}
&:\text{ public and secret key spaces},\\
\mathcal{M}_\lambda,\mathcal{C}_\lambda&:\text{ message and ciphertext spaces},\\
pk,sk&:\text{ public and secret keys},\\
m,ct,m'&:\text{ message, ciphertext, decryption output},\\
\bot&:\text{ decryption failure}.
\end{aligned}
\]
\[
\mathcal{K}_{\mathrm{pub},\lambda},\mathcal{K}_{\mathrm{sec},\lambda},
\mathcal{M}_\lambda,\mathcal{C}_\lambda\subseteq\{0,1\}^{*},
\qquad\bot\notin\{0,1\}^{*}.
\]
\end{definition}

\begin{definition}[Perfect correctness]
\[
\mathrm{PerfectCorrect}(\Pi)\iff
\forall\lambda\in\mathbb{N}\ 
\forall(pk,sk)\in\mathsf{Gen}(1^\lambda)\ 
\forall m\in\mathcal{M}_\lambda:
\]
\[
\Pr\!\left[\mathsf{Dec}(sk,\mathsf{Enc}(pk,m))=m\right]=1.
\]
\end{definition}

\begin{definition}[Perfect security]
\[
\begin{aligned}
\mathrm{PerfectSecure}(\Pi)\iff
&\ \forall\lambda\in\mathbb{N}\;
\forall(pk,sk)\in\mathsf{Gen}(1^\lambda)\\
&\ \forall m_0,m_1\in\mathcal{M}_\lambda\;
\forall ct\in\mathcal{C}_\lambda:\\[3pt]
&\quad |m_0|=|m_1|\ \Longrightarrow\
\Pr_r[\mathsf{Enc}(pk,m_0;r)=ct]
=\Pr_r[\mathsf{Enc}(pk,m_1;r)=ct].
\end{aligned}
\]
\[
\begin{aligned}
\Pi&:\text{ public key encryption scheme},\\
\lambda&:\text{ security parameter},\\
pk,sk&:\text{ generated public and secret keys},\\
\mathcal{M}_\lambda,\mathcal{C}_\lambda&:\text{ message and ciphertext spaces},\\
m_0,m_1&:\text{ messages of equal length},\\
ct&:\text{ ciphertext},\\
|m|&:\text{ bit length of }m,\\
r&:\text{ encryption's random bits},\\
\Pr_r&:\text{ probability over }r.
\end{aligned}
\]
\end{definition}

\begin{definition}[Perfect security: conditional probabilities]
\[
\mathcal{M}_{\lambda,\ell}:=\{m\in\mathcal{M}_\lambda:|m|=\ell\}.
\]
\[
\begin{aligned}
\mathrm{PerfectSecure}_{\mathrm{cond}}(\Pi)\iff
&\ \forall\lambda\in\mathbb{N}\;
\forall(pk,sk)\in\mathsf{Gen}(1^\lambda)\;
\forall\ell\in\mathbb{N}\cup\{0\}\\
&\ \forall D\text{ on }\mathcal{M}_{\lambda,\ell}\;
\forall m\in\mathcal{M}_{\lambda,\ell}\;
\forall ct\in\mathcal{C}_\lambda:\\[2pt]
&\quad\Pr[PK=pk,C=ct]>0\ \Longrightarrow\\
&\qquad\Pr[M=m\mid PK=pk,C=ct]=\Pr[M=m],\\[3pt]
&(PK,SK)\leftarrow\mathsf{Gen}(1^\lambda),\quad
M\leftarrow D,\quad C\leftarrow\mathsf{Enc}(PK,M).
\end{aligned}
\]
\[
\begin{aligned}
\ell&:\text{ fixed message length},\\
\mathcal{M}_{\lambda,\ell}&:\text{ messages of length }\ell,\\
D&:\text{ arbitrary message distribution},\\
PK,SK&:\text{ public and secret key random variables},\\
pk&:\text{ public key value on which we condition},\\
M,C&:\text{ message and ciphertext random variables},\\
\Pr&:\text{ probability over independent key generation, message sampling,}\\
&\phantom{:}\text{ and encryption randomness}.
\end{aligned}
\]
\end{definition}

\begin{lemma}[Equivalence of perfect-security definitions]
\[
\mathrm{PerfectSecure}(\Pi)
\iff\mathrm{PerfectSecure}_{\mathrm{cond}}(\Pi).
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\lambda,\ (pk,sk),\ \ell&:\text{ fixed allowed parameter, generated keys, and length},\\
\Pr_{pk}[E]&:=\Pr[E\mid PK=pk],\\
q_m(ct)&:=\Pr_r[\mathsf{Enc}(pk,m;r)=ct],\\
p_m&:=\Pr[M=m]=\Pr_{pk}[M=m]\qquad(M\perp PK).
\end{aligned}
\]
\[
\begin{aligned}
(\Longrightarrow)\quad
&q_m(ct)=q(ct)\quad\forall m\in\mathcal{M}_{\lambda,\ell}\\
&\Longrightarrow\quad
\Pr_{pk}[C=ct]=\sum_{u\in\mathcal{M}_{\lambda,\ell}}p_uq_u(ct)=q(ct)\\
&\Longrightarrow\quad
\Pr_{pk}[M=m\mid C=ct]
=\frac{p_mq_m(ct)}{\Pr_{pk}[C=ct]}=p_m
\quad(\Pr_{pk}[C=ct]>0).
\end{aligned}
\]
\[
\begin{aligned}
(\Longleftarrow)\quad
&m_0\ne m_1,\quad |m_0|=|m_1|=\ell,\quad
D:=\mathrm{Uniform}(\{m_0,m_1\}),\\
&s:=q_{m_0}(ct)+q_{m_1}(ct),\qquad \Pr_{pk}[C=ct]=s/2,\\
&s>0\ \Longrightarrow\
\frac{q_{m_0}(ct)}{s}
=\Pr_{pk}[M=m_0\mid C=ct]=\frac12
\ \Longrightarrow\ q_{m_0}(ct)=q_{m_1}(ct),\\
&s=0\ \Longrightarrow\ q_{m_0}(ct)=q_{m_1}(ct)=0.
\end{aligned}
\]
\end{proof}

\begin{problem}[Exercise sheet 1: Perfect Secrecy]
Prove that perfect secrecy is impossible for PKE schemes.
\end{problem}
\[
\begin{aligned}
\mathrm{PerfectCorrect}(\Pi),\quad
\lambda\in\mathbb{N},\quad (pk,sk)\in\mathsf{Gen}(1^\lambda),\\
m_0,m_1\in\mathcal{M}_\lambda,\quad m_0\ne m_1,\quad |m_0|=|m_1|=\ell.
\end{aligned}
\]
\begin{proof}[Solution 1: ciphertext distributions]
\[
q_i(ct):=\Pr_r[\mathsf{Enc}(pk,m_i;r)=ct]\qquad(i\in\{0,1\}).
\]
\[
\begin{aligned}
&\exists ct\in\mathcal{C}_\lambda:\\
&\quad q_0(ct)>0\ \Longrightarrow\quad
\Pr_s[\mathsf{Dec}(sk,ct;s)=m_0]=1,\\
&\quad q_1(ct)>0\ \Longrightarrow\
\Pr_s[\mathsf{Dec}(sk,ct;s)=m_1]=1,\\
&\quad m_0\ne m_1\ \Longrightarrow\
\Pr_s[\mathsf{Dec}(sk,ct;s)=m_0]
+\Pr_s[\mathsf{Dec}(sk,ct;s)=m_1]\le1\\
&\quad\Longrightarrow\quad
\bigl(q_1(ct)=0\land q_0(ct)>0\bigr)
\lor\bigl(q_1(ct)>0\land q_0(ct)=0\bigr)
\ \Longrightarrow\ q_1(ct)\ne q_0(ct)\\
&\quad\Longrightarrow\quad\neg\mathrm{PerfectSecure}(\Pi).
\end{aligned}
\]
\[
r,s:\text{ encryption and decryption random bits}.
\]
\end{proof}

\begin{samepage}
\begin{proof}[Solution 2: conditional probabilities]
\[
\begin{aligned}
\mathsf{Find}_\lambda(pk):\quad
&r\in\{0,1\}^{*}\text{ enumerated by increasing length}:\\
&\mathsf{Gen}(1^\lambda;r)=(pk,sk')\ \Longrightarrow\ \text{return }sk',\\
A_\lambda(pk,ct)&:=\mathsf{Dec}(\mathsf{Find}_\lambda(pk),ct).
\end{aligned}
\]
\[
\begin{aligned}
(pk,sk)\in\mathsf{Gen}(1^\lambda)
&\Longrightarrow\ \mathsf{Find}_\lambda(pk)\text{ terminates},\\
&\Longrightarrow\ (pk,\mathsf{Find}_\lambda(pk))\in\mathsf{Gen}(1^\lambda),\\
&\Longrightarrow\ \Pr[A_\lambda(pk,\mathsf{Enc}(pk,m))=m]=1.
\end{aligned}
\]
\[
\begin{aligned}
&D:=\mathrm{Uniform}(\{m_0,m_1\}),\quad
(PK,SK)\leftarrow\mathsf{Gen}(1^\lambda),\quad
M\leftarrow D,\quad C\leftarrow\mathsf{Enc}(PK,M),\\
&\Pr[A_\lambda(PK,C)=M]=1,\qquad \Pr[M=m_0]=\tfrac12,\\
&ct:\quad\Pr_r[\mathsf{Enc}(pk,m_0;r)=ct]>0\\
&\Longrightarrow\quad\Pr[PK=pk,C=ct]>0,\\
&\Pr[A_\lambda(pk,ct)=m_0]=1\\
&\Longrightarrow\quad\Pr[M=m_0\mid PK=pk,C=ct]=1\ne\tfrac12=\Pr[M=m_0]\\
&\Longrightarrow\quad\neg\mathrm{PerfectSecure}_{\mathrm{cond}}(\Pi).
\end{aligned}
\]
\[
\begin{aligned}
\mathsf{Find}_\lambda&:\text{ exhaustive search for a compatible secret key},\\
A_\lambda&:\text{ attacker with no polynomial runtime restriction},\\
D&:\text{ message distribution independent of key generation}.
\end{aligned}
\]
\end{proof}
\end{samepage}


\begin{definition}[Semantic security]
\[
\begin{aligned}
\mathrm{SemanticSecure}(\Pi)\iff
&\ \forall\text{ PPT }A\;\exists\text{ PPT }A'
\;\forall\text{ PPT }\mathsf{Samp}\;\forall\text{ PT }f:\\
&\quad \forall\lambda\in\mathbb{N}:\quad
|\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)|\leq\negl(\lambda).
\end{aligned}
\]
\[
\begin{aligned}
\epsilon_A(\lambda)&:=\Pr\!\left[
\begin{array}{l}
(pk,sk)\leftarrow\mathsf{Gen}(1^\lambda)\\
(m,h)\leftarrow\mathsf{Samp}(pk)
\end{array}
\ :\ A(pk,h,\mathsf{Enc}(pk,m))=f(m,h)
\right],\\[5pt]
\epsilon_{A'}(\lambda)&:=\Pr\!\left[
\begin{array}{l}
(pk,sk)\leftarrow\mathsf{Gen}(1^\lambda)\\
(m,h)\leftarrow\mathsf{Samp}(pk)
\end{array}
\ :\ A'(pk,h,|m|)=f(m,h)
\right].
\end{aligned}
\]
\[
\begin{aligned}
\Pi&:\text{ public key encryption scheme},\\
A&:\text{ attacker given the public key, hint, and ciphertext},\\
A'&:\text{ simulator given the public key, hint, and message length},\\
\mathsf{Samp}&:\text{ sampler returning }(m,h)\text{ with }m\in\mathcal{M}_\lambda,\\
h&:\text{ hint containing public information about }m,\\
f&:\text{ deterministic polynomial-time function of }(m,h),\\
\epsilon_A,\epsilon_{A'}&:\text{ probabilities of correctly computing }f(m,h),\\
\negl&:\text{ negligible bound},\\
\lambda&:\text{ security parameter},\\
\Pr&:\text{ probability over all random bits in the experiment}.
\end{aligned}
\]
\end{definition}

\begin{theorem}[Semantic security $\iff$ IND-CPA]
\[
\mathrm{SemanticSecure}(\Pi)
\iff\mathrm{IND\text{-}CPA\text{-}Secure}(\Pi).
\]
\[
\begin{aligned}
\Pi&:\text{ public key encryption scheme}.
\end{aligned}
\]
\end{theorem}

\begin{proof}[First direction: semantic security $\Longrightarrow$ IND-CPA]
\[
\begin{aligned}
\text{Contrapositive goal:}\quad
&\left[\begin{gathered}
\exists\text{ PPT }D:\quad
\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}\text{ non-negligible}
\end{gathered}\right]\\
&\quad\Longrightarrow\quad
\left[\begin{gathered}
\exists\text{ PPT }A\;\forall\text{ PPT }A'\;
\exists\text{ PPT }\mathsf{Samp}\;\exists\text{ PT }f:\\
|\epsilon_A-\epsilon_{A'}|\text{ non-negligible}
\end{gathered}\right].
\end{aligned}
\]
\[
\begin{aligned}
\text{Idea:}\quad
&\left[\begin{gathered}
\exists\text{ PPT }A\;\exists\text{ PPT }\mathsf{Samp}\;
\exists\text{ PT }f\;\forall\text{ PPT }A':\\
\exists\text{ polynomial }p\geq1\;
\forall\lambda\in\mathbb{N}:\\
|\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)|
\geq\frac{\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}(\lambda)}{p(\lambda)}
-\negl(\lambda)
\end{gathered}\right]\\
&\quad\overset{\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}\text{ non-negligible}}{\Longrightarrow}\quad
\forall\text{ PPT }A':\quad
|\epsilon_A-\epsilon_{A'}|\text{ non-negligible}.
\end{aligned}
\]
\[
\neg\mathrm{IND\text{-}CPA\text{-}Secure}(\Pi)
\ \Longrightarrow\
\exists\text{ PPT }D=(D_0,D_1):\quad
\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}\text{ non-negligible}.
\]
\[
\begin{aligned}
&D^{(0)}:=(D_0,D_1),\qquad D^{(1)}:=(D_0,1-D_1),\\
&P_{\Pi,D^{(1)},\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)=1-P_{\Pi,D^{(0)},\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda),\\
&\max_{i\in\{0,1\}}P_{\Pi,D^{(i)},\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)
=\frac12+\frac12\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}(\lambda),\\
&\Longrightarrow\quad
\exists i\in\{0,1\}:\\
&\qquad P_{\Pi,D^{(i)},\mathrm{win}}^{\mathrm{IND\text{-}CPA}}-\frac12
\text{ non-negligibly positive},\\
&D:=D^{(i)}.
\end{aligned}
\tag{1}\label{eq:indcpa-positive-gap}
\]
\[
\begin{aligned}
\mathsf{Samp}(pk):\quad
&(m_0,m_1,st)\leftarrow D_0(pk);\\
&b\leftarrow_{\$}\{0,1\};\\
&\mathrm{return}\ (m_b,(m_0,m_1,st));\\[4pt]
f(m_b,(m_0,m_1,st))&:=\mathbf{1}[m_b\ne m_0]=b;\\[4pt]
A(pk,(m_0,m_1,st),ct)&:=D_1(st,ct).
\end{aligned}
\]
\[
\begin{aligned}
\epsilon_A(\lambda)
&=\Pr\!\left[
\begin{array}{l}
(pk,sk)\leftarrow\mathsf{Gen}(1^\lambda)\\
(m_b,h)\leftarrow\mathsf{Samp}(pk)
\end{array}
\ :\ A(pk,h,\mathsf{Enc}(pk,m_b))=f(m_b,h)\right]\\[6pt]
&=\Pr\!\left[
\begin{array}{l}
(pk,sk)\leftarrow\mathsf{Gen}(1^\lambda)\\
(m_0,m_1,st)\leftarrow D_0(pk)\\
b\leftarrow_{\$}\{0,1\}
\end{array}
\ :\ D_1(st,\mathsf{Enc}(pk,m_b))=b\right]\\[6pt]
&=P_{\Pi,D,\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda).
\end{aligned}
\]
\[
\begin{aligned}
&\forall\text{ PPT }A':\quad
b\text{ independent of }(pk,h,|m_b|),\\
&\qquad Y:=A'(pk,h,|m_b|),\\
&\qquad\epsilon_{A'}(\lambda)=\Pr[Y=b]\\
&\qquad\quad=\Pr[b=0\land Y=0]+\Pr[b=1\land Y=1]\\
&\qquad\quad=\Pr[b=0]\Pr[Y=0]+\Pr[b=1]\Pr[Y=1]\\
&\qquad\quad=\frac12\Pr[Y=0]+\frac12\Pr[Y=1]\\
&\qquad\quad=\frac12\bigl(\Pr[Y=0]+\Pr[Y=1]\bigr)
\leq\frac12\cdot1=\frac12,\\
&\qquad|\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)|
\geq P_{\Pi,D,\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)-\frac12,\\
&\overset{\eqref{eq:indcpa-positive-gap}}{\Longrightarrow}\quad
\exists\text{ PPT }A\;\forall\text{ PPT }A'\;
\exists\text{ PPT }\mathsf{Samp}\;\exists\text{ PT }f:\\
&\qquad|\epsilon_A-\epsilon_{A'}|\text{ non-negligible}\\
&\Longrightarrow\quad\neg\mathrm{SemanticSecure}(\Pi).
\end{aligned}
\]
\[
\begin{aligned}
D&:\text{ IND-CPA attacker, with its guess flipped if needed},\\
P_{\Pi,D,\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)&:\text{ winning probability of }D,\\
\mathsf{Samp}&:\text{ sampler choosing uniformly between }D\text{'s messages},\\
h=(m_0,m_1,st)&:\text{ hint independent of the hidden bit }b,\\
f&:\text{ function identifying which distinct message was sampled},\\
A&:\text{ semantic attacker running }D_1,\\
A'&:\text{ arbitrary simulator given only }pk,h,|m_b|.
\end{aligned}
\]
\end{proof}

\begin{proof}[By direct proof: IND-CPA $\Longrightarrow$ semantic security]
\leavevmode\par\smallskip\hrule\medskip
\[
\begin{aligned}
\text{Goal:}\quad
&\left[\begin{gathered}
\forall\text{ PPT }D\;
\forall\lambda\in\mathbb{N}:\\
\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}(\lambda)\leq\negl(\lambda)
\end{gathered}\right]\\
&\quad\Longrightarrow\quad
\left[\begin{gathered}
\forall\text{ PPT }A\;\exists\text{ PPT }A'\;
\forall\text{ PPT }\mathsf{Samp}\;\forall\text{ PT }f:\\
\forall\lambda\in\mathbb{N}:\quad
|\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)|\leq\negl(\lambda)
\end{gathered}\right].
\end{aligned}
\]
\[
\begin{aligned}
\text{Idea:}\quad
&\left[\begin{gathered}
\exists\text{ PPT }D\;\exists\text{ polynomial }p\;
\forall\lambda\in\mathbb{N}:\\
|\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)|
\leq p(\lambda)\,\mathrm{Adv}^{\mathrm{IND\text{-}CPA}}_{\Pi,D}(\lambda)
+\negl(\lambda)
\end{gathered}\right]\\
&\quad\overset{\mathrm{IND\text{-}CPA\text{-}Secure}(\Pi)}{\Longrightarrow}\quad
|\epsilon_A-\epsilon_{A'}|\text{ negligible}.
\end{aligned}
\]
\renewcommand{\qedsymbol}{}
\[
\forall\text{ PPT }D:\quad
\underbrace{2\left|P_{\Pi,D,\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)-\frac12\right|}_{\mathrm{Adv}_{\Pi,D}^{\mathrm{IND\text{-}CPA}}(\lambda)}
\overset{\mathrm{IND\text{-}CPA\text{-}Secure}(\Pi)}{\leq}\negl(\lambda).
\tag{2}\label{eq:indcpa-negligible-bound}
\]
\[
\text{Idea 2:}\quad\text{Construct the IND-CPA game for Idea 1.}
\]
We want to construct an IND-CPA attacker $D$ so that both
$\epsilon_A$ and $\epsilon_{A'}$ appear in its winning probability.
This would let us relate their difference to $D$'s advantage, as required
by Idea 1. The IND-CPA game already provides a challenge bit $b$ with
two possible values. We use these two branches to include both executions
in the same game. When $b=0$, this runs $A$ on an
encryption of $m$, giving the real execution. When $b=1$, it runs $A$ on
an encryption of $0^{|m|}$, which is exactly how $A'$ is defined below.
Thus the probabilities of correctly computing $f(m,h)$ in the two branches
are $\epsilon_A$ and $\epsilon_{A'}$. We can then use them to calculate
$D$'s winning probability and advantage in this one IND-CPA game,
and thereby bound the semantic-security gap.
\[
\forall\text{ PPT }A:\quad
A'(pk,h,\ell):=A(pk,h,\mathsf{Enc}(pk,0^\ell)).
\]
\[
\begin{aligned}
A'&\text{ depends only on }A\text{ and }\Pi,\\
\forall\text{ PPT }\mathsf{Samp}\;\forall\text{ PT }f:\quad
D&=(D')^A\text{ uses the chosen }\mathsf{Samp}\text{ and }f.
\end{aligned}
\]
\par\pagebreak[0]
\begin{samepage}
\begingroup\small
\[
\begin{array}{c@{\;}c@{\;}|c@{\;}c@{\;}c|}
\multicolumn{2}{c}{\text{IND-CPA challenger}}&\multicolumn{3}{c}{D=(D')^A}\\[3pt]
\cline{3-5}
&&\rule{0pt}{18pt}D'&&A\\[5pt]
\cline{3-5}

&\xrightarrow{\ pk\ }&&&\\
&&(m,h)\leftarrow\mathsf{Samp}(pk)&&\\

&&m_0\ne m_1&&\\
&\xleftarrow{\ m_0,m_1\ }&&&\\
\cline{3-5}
&&\multicolumn{3}{|c|}{\rule{0pt}{14pt}D_2}\\[3pt]
\cline{3-5}

&&&\xrightarrow{\ pk,h,ct\ }&\\

\[
\begin{aligned}
\Pr[D\text{ wins}\mid b=0]
&=\Pr[A(pk,h,\mathsf{Enc}(pk,m))=f(m,h)]\\
&=\epsilon_A(\lambda),\\[4pt]
\Pr[D\text{ wins}\mid b=1]
&=\Pr[A'(pk,h,|m|)\ne f(m,h)]\\
&=1-\Pr[A'(pk,h,|m|)=f(m,h)]\\
&=1-\epsilon_{A'}(\lambda).
\end{aligned}
\]
\[
\begin{aligned}
P_{\Pi,D,\mathrm{win}}^{\mathrm{IND\text{-}CPA}}(\lambda)
&=\Pr[D\text{ wins}\mid b=0]\Pr[b=0]\\
&\quad+\Pr[D\text{ wins}\mid b=1]\Pr[b=1]\\
&=\frac12\epsilon_A(\lambda)+\frac12(1-\epsilon_{A'}(\lambda))\\
&=\frac12+\frac12\bigl(\epsilon_A(\lambda)-\epsilon_{A'}(\lambda)\bigr).
\end{aligned}
\]

&\overset{\eqref{eq:indcpa-negligible-bound}}{\leq}\negl(\lambda).
\end{aligned}
$}

\quad\text{non-negligible}\\
&\overset{\mathrm{IND\text{-}CPA\text{-}Secure}(\Pi)}{\Longrightarrow}\bot.
\end{aligned}

$}

\end{proof}

\subsection*{Questions}
\begin{enumerate}
\item Is $|m_0|=|m_1|$ a standing assumption throughout the lectures
whenever two messages are compared, or must this restriction be stated
explicitly in each definition where it applies?
If so, does this also apply to perfect security?
\item Would it be better to allow arbitrary PPT attackers and have the
IND-CPA challenger explicitly check that $m_0,m_1\in\mathcal{M}_\lambda$,
$m_0\ne m_1$, and $|m_0|=|m_1|$, rather than restrict to admissible
attackers? If a check fails, could the game return a uniformly random
bit, so the winning probability is $1/2$ and the advantage is zero?
\item Does it make sense to include explicit public key, secret key,
and ciphertext spaces
$\mathcal{K}_{\mathrm{pub},\lambda}$,
$\mathcal{K}_{\mathrm{sec},\lambda}$, and $\mathcal{C}_\lambda$
in the PKE definition, as written above, even though the lecture
specifies only the message space?
\item The lecture uses both right-pointing and left-pointing arrows,
but its initial notation definition specifies the left-pointing arrow
for sampling algorithm outputs. In our PKE definition, we write
$\mathsf{Gen}(1^\lambda)\longrightarrow(pk,sk)$.
Is this consistent with that notation definition, or should we write
$(pk,sk)\leftarrow\mathsf{Gen}(1^\lambda)$ instead?
\item The exercise solutions define perfect secrecy by
$\Pr[M=m\mid C=c]=\Pr[M=m]$.
For PKE, should the public key also be included in the conditioning,
since the attacker observes both $pk$ and $ct$?
\end{enumerate}
